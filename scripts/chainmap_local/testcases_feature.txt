# chainmap local-constraint feature test cases
# Format: name workdir(relative to data/) command... | assert1;assert2;...
# Assert types:
#   contains:TEXT        - output must contain TEXT
#   not_contains:TEXT    - output must NOT contain TEXT
#   mapping:C1->C2,...   - chain1 list must pair to chain2 list in order

# ---- G2/G3: partial mapping - mapping fixed + auto selection ----
# (pairing-summary "Protein: N pair(s)" asserted in G7_summary)
F1_partial_mapping    . complexA.pdb complexB.pdb -mm 1 -ter 0 -chainmap map_A.txt -outfmt 2 | mapping:A->A,B->B,C->C
F2_two_mappings       . complexA.pdb complexB.pdb -mm 1 -ter 0 -chainmap map_AB.txt -outfmt 2 | mapping:A->A,B->B,C->C

# ---- G7: pairing summary format ----
G7_summary            . complexA.pdb complexB.pdb -mm 1 -ter 0 -chainmap map_AB.txt -outfmt 2 | contains:Chain pairing summary:;contains:Protein: 3 pair(s) aligned;contains:RNA: 0 pair(s) aligned

# ---- G6.1: duplicate target defense ----
# 提示需带出完整上下文：目标链所属结构、占用它的映射、被忽略的条目
F3_duplicate_target   . complexA.pdb complexB.pdb -mm 1 -ter 0 -chainmap map_dup.txt -outfmt 2 | contains:chain A of complexB.pdb is already the target of mapping;contains:mapping C -> A is ignored

# ---- G2.3: crossed mapping lock (4iaj A -> 4jhm B, even if auto would pick differently) ----
F4_lock_conflict      . 4iaj.pdb1 4jhm.pdb1 -mm 1 -ter 0 -chainmap map_4iaj_AtoB.txt -outfmt 2 | mapping:A->B

# ---- G5.2: chain number mismatch ----
E1_chain_mismatch     . complexA.pdb complexB_2chain.pdb -mm 1 -ter 0 -outfmt 2 | contains:Protein: 2 pair(s) aligned;contains:more chains in structure 1

# ---- G5.1: molecule type mismatch (auto) ----
E2_type_mismatch      . complexA_plusRNA.pdb complexB.pdb -mm 1 -ter 0 -outfmt 2 | contains:Protein: 3 pair(s) aligned;contains:no chain of the same molecule type;contains:(RNA)

# ---- G5.1: -mol protein pre-scan (all-RNA input, no usable chain) ----
# 预检失败无结果行，表头（#PDBchain1）也不应输出
E3_mol_prescan        . ../../mm1/data/MSTATest/US7351924051.pdb complexA.pdb -mm 1 -ter 0 -mol protein -outfmt 2 | contains:US7351924051.pdb contains 1 chain(s), all RNA, but -mol protein is set;not_contains:0 chain;not_contains:no CA atoms;not_contains:\US7351924051.pdb;not_contains:/US7351924051.pdb;not_contains:#PDBchain1

# ---- G6.3: type-mismatch mapping - warning + ignore ----
# 全部映射无效报错退出，无结果行，表头也不应输出
E4_mapping_type_mismatch . ../../mm1/data/MSTATest/US7351924051.pdb complexA.pdb -mm 1 -ter 0 -chainmap map_rna_type.txt -outfmt 2 | contains:molecule type mismatch;not_contains:#PDBchain1

# ---- G8.3: partial mapping + chain mismatch ----
B2_mapping_plus_mismatch . complexA.pdb complexB_2chain.pdb -mm 1 -ter 0 -chainmap map_A.txt -outfmt 2 | contains:Protein: 2 pair(s) aligned;contains:more chains in structure 1;mapping:A->A

# ---- G6.4: nonexistent chain id in chainmap ----
B4_nonexistent_chain . complexA.pdb complexB.pdb -mm 1 -ter 0 -chainmap map_nonexist.txt -outfmt 2 | contains:Chainmap: 1 entries specified, 0 mapped, 3 free-matching; total 3 pair(s) aligned;contains:Invalid mappings: X -> A (chain X does not exist in structure 1

# ---- 其它输出方式验证：-outfmt 0（默认详细格式，配对汇总无条件输出）----
G8_outfmt0_detail   . complexA.pdb complexB.pdb -mm 1 -ter 0 -chainmap map_A.txt | contains:Name of Structure_1:;contains:TM-score=;contains:Chain pairing summary:;contains:Chainmap: 1 entries specified, 1 mapped, 2 free-matching

# ---- 其它输出方式验证：-outfmt 1（fasta 格式）----
G9_outfmt1_fasta    . complexA.pdb complexB.pdb -mm 1 -ter 0 -chainmap map_A.txt -outfmt 1 | contains:>complexA.pdb:1,A:1,B:1,C;contains:# Lali=;contains:Chain pairing summary:;contains:Protein: 3 pair(s) aligned

# ---- 其它输出方式验证：-m -（旋转矩阵输出到 stdout，与配对汇总共存）----
G10_matrix_stdout   . complexA.pdb complexB.pdb -mm 1 -ter 0 -chainmap map_A.txt -outfmt 2 -m - | contains:The rotation matrix to rotate Structure_1 to Structure_2;contains:X[i] = t[0] + u[0][0]*x[i];contains:Chain pairing summary:;mapping:A->A

# ---- G11: cross-chain adoption critical zone（回归 MMalign_dimer 双重计数缺陷）----
# crit_A/B 合成数据：B 链 y 偏移差 8A → 真实 cross ∈ [iteration/2, iteration]
# 修复版：不采纳（配 B->B，A 链质量保护剔除）；缺陷版：误采纳（配 A->A，B 链剔除）
G11_cross_critical   . crit_A.pdb crit_B.pdb -mm 1 -ter 0 -outfmt 2 | mapping:B->B;contains:Protein: 1 pair(s) aligned
